firestore.rules (Role-Enforced Security Rules)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAuthenticated() {
      return request.auth != null;
    }
    function adminDoc(appId) {
      return get(/databases/$(database)/documents/artifacts/$(appId)/public/data/administrators/$(request.auth.uid)).data;
    }
    function isActiveAdmin(appId) {
      return isAuthenticated() && adminDoc(appId).accountStatus == 'Active';
    }
    function isCommand(appId) {
      return isActiveAdmin(appId) && adminDoc(appId).accessLevel == 'Battalion Command';
    }
    function isCommandOrLogistics(appId) {
      return isActiveAdmin(appId) && (adminDoc(appId).accessLevel == 'Battalion Command' || adminDoc(appId).accessLevel == 'Battalion Logistics');
    }

    match /artifacts/{appId}/public/data/administrators/{adminId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isCommand(appId);
    }
    match /artifacts/{appId}/public/data/feCodes/{codeId} {
      allow read, create, update: if isActiveAdmin(appId);
      allow delete: if isCommandOrLogistics(appId);
    }
    match /artifacts/{appId}/public/data/instructorLicenses/{licId} {
      allow read: if isCommandOrLogistics(appId) || (isActiveAdmin(appId) && resource.data.linkedUsername == adminDoc(appId).username);
      allow create, update, delete: if isCommandOrLogistics(appId);
    }
    match /artifacts/{appId}/public/data/licenseRenewalRequests/{reqId} {
      allow read: if isCommandOrLogistics(appId) || (isActiveAdmin(appId) && resource.data.instructorUsername == adminDoc(appId).username);
      allow create: if isActiveAdmin(appId);
      allow update: if isCommandOrLogistics(appId);
    }
    match /artifacts/{appId}/public/data/auditLogs/{logId} {
      allow read: if isCommand(appId);
      allow create: if isAuthenticated();
      allow update, delete: if false;
    }
  }
}
